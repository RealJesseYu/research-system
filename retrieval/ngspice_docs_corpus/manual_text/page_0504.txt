Generates a temperature vector
proc t e m p e r a t u r e s _ c a l c { temp_inf temp_sup points } {
set tstep [ expr " ( $temp_sup - $temp_inf ) / $points " ]
set t $temp_inf
set t e m p e r a t u r e s ""
for { set i 0 } { $i < $points } { incr i } {
set t [ expr { $t + $tstep } ]
set t e m p e r a t u r e s " $ t e m p e r a t u r e s $t "
}
return $ t e m p e r a t u r e s }
